const samples = InParam && InParam.samples;
if (!Array.isArray(samples) || samples.length !== 5) {
  OutParam.status = "need_five_samples";
} else if (!samples.every(sample => sample && sample.Q === 192 && Number.isFinite(sample.V))) {
  OutParam.status = "bad_sample";
} else {
  OutParam.status = "ok";
  OutParam.mean = samples.reduce((sum, sample) => sum + sample.V, 0) / samples.length;
}
endResponse(RequestID);
